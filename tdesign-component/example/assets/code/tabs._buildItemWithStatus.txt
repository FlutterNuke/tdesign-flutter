
  Widget _buildItemWithStatus(BuildContext context) {
    var tabs = [
      const TTab(
        text: '选中',
      ),
      const TTab(
        text: '默认',
      ),
      const TTab(
        text: '禁用',
        enable: false,
      ),
    ];
    return TTabBar(
        tabs: tabs,
        controller: TabController(length: 3, vsync: this),
        backgroundColor: Colors.white,
        showIndicator: true);
  }
