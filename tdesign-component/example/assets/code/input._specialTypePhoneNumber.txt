
  Widget _specialTypePhoneNumber(BuildContext context) {
    return Column(
      children: [
        TInput(
          type: TInputType.normal,
          controller: controller[14],
          leftLabel: '手机号',
          hintText: '输入手机号',
          backgroundColor: Colors.white,
          rightBtn: SizedBox(
            width: 98,
            child: Row(
              mainAxisAlignment: MainAxisAlignment.spaceBetween,
              children: [
                Padding(
                  padding: const EdgeInsets.only(right: 16),
                  child: Container(
                    width: 0.5,
                    height: 24,
                    color: TTheme.of(context).grayColor3,
                  ),
                ),
                _countdownTime > 0
                    ? TText(
                        '${countDownText}(${_countdownTime}秒)',
                        textColor: TTheme.of(context).fontGyColor4,
                      )
                    : TText(confirmText, textColor: TTheme.of(context).brandNormalColor),
              ],
            ),
          ),
          needClear: false,
          onBtnTap: () {
            if (_countdownTime == 0) {
              TToast.showText('点击了发送验证码', context: context);
              setState(() {
                _countdownTime = 60;
              });
              startCountdownTimer();
            }
          },
        ),
        const SizedBox(
          height: 16,
        )
      ],
    );
  }
