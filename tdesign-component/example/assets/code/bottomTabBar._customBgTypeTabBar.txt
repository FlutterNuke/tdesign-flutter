
  Widget _customBgTypeTabBar(BuildContext context) {
    return TBottomTabBar(TBottomTabBarBasicType.text,
        backgroundColor: TTheme.of(context).successColor6,
        selectedBgColor: TTheme.of(context).errorColor1,
        unselectedBgColor: TTheme.of(context).brandColor1,
        useVerticalDivider: false,
        navigationTabs: [
          TBottomTabBarTabConfig(
            tabText: '标签',
            onTap: () {
              onTapTab(context, '标签1');
            },
          ),
          TBottomTabBarTabConfig(
            tabText: '标签',
            unselectTabTextStyle:
                TextStyle(color: TTheme.of(context).fontGyColor1),
            onTap: () {
              onTapTab(context, '标签1');
            },
          ),
        ]);
  }
