
  Widget _buildShadowsMiddle(BuildContext context) {
    return Container(
      width: 100,
      height: 50,
      decoration: BoxDecoration(
          color: TTheme.of(context).whiteColor1,
          boxShadow: TTheme.of(context).shadowsMiddle,
          borderRadius:
              BorderRadius.circular(TTheme.of(context).radiusDefault)),
    );
  }
