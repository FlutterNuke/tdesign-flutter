
TTimeCounter _buildCustomUnitSmallSize(BuildContext context) {
  var style = TTimeCounterStyle.generateStyle(context, size: TTimeCounterSize.small);
  style.timeColor = TTheme.of(context).errorColor6;
  return TTimeCounter(
    time: 60 * 60 * 1000,
    splitWithUnit: true,
    style: style,
  );
}
