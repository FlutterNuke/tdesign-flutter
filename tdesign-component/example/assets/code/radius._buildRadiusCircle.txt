
  Widget _buildRadiusCircle(BuildContext context) {
    //  圆形与胶囊型一致，如果长款一致即是圆形
    return Container(
      width: 50,
      height: 50,
      decoration: BoxDecoration(
          color: TTheme.of(context).brandNormalColor,
          borderRadius:
              BorderRadius.circular(TTheme.of(context).radiusCircle)),
    );
  }
